### Q17 (wrong_result): Show sessions and bounce rate for each landing page.

The landing page is the first pageview of the session.

```sql
WITH landing_pages AS (
    SELECT
        ws.website_session_id,
        wp.pageview_url AS landing_page
    FROM toystore.website_sessions ws
    JOIN (
        SELECT
            website_session_id,
            pageview_url,
            ROW_NUMBER() OVER (PARTITION BY website_session_id ORDER BY created_at ASC, website_pageview_id ASC) AS rn
        FROM toystore.website_pageviews
    ) wp ON ws.website_session_id = wp.website_session_id AND wp.rn = 1
),
session_counts AS (
    SELECT
        website_session_id,
        COUNT(*) AS pageview_count
    FROM toystore.website_pageviews
    GROUP BY website_session_id
)
SELECT
    lp.landing_page,
    COUNT(lp.website_session_id) AS total_sessions,
    SUM(CASE WHEN sc.pageview_count = 1 THEN 1 ELSE 0 END)::NUMERIC / COUNT(lp.website_session_id) AS bounce_rate
FROM landing_pages lp
JOIN session_counts sc ON lp.website_session_id = sc.website_session_id
GROUP BY lp.landing_page
ORDER BY total_sessions DESC
```

### Q19 (wrong_result): For each month since the first cross-sell item was sold, what percentage of orders had more than one item?

The start date must come from the data, and the last month is partial.

```sql
WITH first_cross_sell AS (
    SELECT MIN(created_at) AS first_date
    FROM toystore.order_items
    WHERE is_primary_item = 0
),
monthly_orders AS (
    SELECT 
        DATE_TRUNC('month', o.created_at)::DATE AS order_month,
        COUNT(o.order_id) AS total_orders,
        COUNT(CASE WHEN o.items_purchased > 1 THEN o.order_id END) AS orders_with_multiple_items
    FROM toystore.orders o
    CROSS JOIN first_cross_sell f
    WHERE o.created_at >= f.first_date
    GROUP BY DATE_TRUNC('month', o.created_at)
)
SELECT 
    order_month,
    total_orders,
    orders_with_multiple_items,
    ROUND(100.0 * orders_with_multiple_items / NULLIF(total_orders, 0), 2) AS percentage_multiple_items
FROM monthly_orders
ORDER BY order_month
```
