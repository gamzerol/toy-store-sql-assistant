### Q18 (wrong_result): Compare the order conversion rate of /billing and /billing-2 during the period when both pages were live.

Needs the overlap window. Comparing all time would be unfair.

```sql
WITH billing_test_dates AS (
  SELECT
    MIN(created_at) AS start_at,
    MAX(created_at) AS end_at
  FROM toystore.website_pageviews
  WHERE pageview_url = '/billing-2'
),
session_billing_pages AS (
  SELECT
    pv.website_session_id,
    MAX(CASE WHEN pv.pageview_url = '/billing' THEN 1 ELSE 0 END) AS viewed_billing,
    MAX(CASE WHEN pv.pageview_url = '/billing-2' THEN 1 ELSE 0 END) AS viewed_billing_2
  FROM toystore.website_pageviews pv
  JOIN billing_test_dates d
    ON pv.created_at >= d.start_at AND pv.created_at <= d.end_at
  WHERE pv.pageview_url IN ('/billing', '/billing-2')
  GROUP BY pv.website_session_id
),
session_page_type AS (
  SELECT
    website_session_id,
    CASE
      WHEN viewed_billing_2 = 1 AND viewed_billing = 0 THEN '/billing-2'
      WHEN viewed_billing = 1 AND viewed_billing_2 = 0 THEN '/billing'
      ELSE NULL
    END AS billing_page_version
  FROM session_billing_pages
),
session_orders AS (
  SELECT
    website_session_id,
    COUNT(order_id) AS order_count
  FROM toystore.orders
  GROUP BY website_session_id
)
SELECT
  pt.billing_page_version,
  COUNT(DISTINCT pt.website_session_id) AS sessions,
  COUNT(DISTINCT o.order_id) AS orders,
  COUNT(DISTINCT o.order_id)::NUMERIC / NULLIF(COUNT(DISTINCT pt.website_session_id), 0) AS conversion_rate
FROM session_page_type pt
LEFT JOIN toystore.orders o
  ON pt.website_session_id = o.website_session_id
WHERE pt.billing_page_version IS NOT NULL
GROUP BY pt.billing_page_version
```

### Q19 (wrong_result): For each month since the first cross-sell item was sold, what percentage of orders had more than one item?

The start date must come from the data, and the last month is partial.

```sql
SELECT DATE_TRUNC('month', o.created_at)::DATE AS month, COUNT(DISTINCT o.order_id) AS total_orders, COUNT(DISTINCT CASE WHEN o.items_purchased > 1 THEN o.order_id END) AS multi_item_orders, ROUND(COUNT(DISTINCT CASE WHEN o.items_purchased > 1 THEN o.order_id END)::NUMERIC / NULLIF(COUNT(DISTINCT o.order_id), 0) * 100, 2) AS percentage_multi_item FROM toystore.orders o JOIN toystore.order_items oi ON o.order_id = oi.order_id WHERE oi.is_primary_item = 0 AND o.created_at >= (SELECT MIN(created_at) FROM toystore.order_items WHERE is_primary_item = 0) GROUP BY DATE_TRUNC('month', o.created_at) ORDER BY month
```

### Q25 (wrong_result): Compare the conversion rate of new and repeat sessions across all time.

New vs repeat comes from is_repeat_session.

```sql
SELECT CASE WHEN s.is_repeat_session = 1 THEN 'repeat' ELSE 'new' END AS session_type, COUNT(DISTINCT s.website_session_id) AS total_sessions, COUNT(DISTINCT o.order_id) AS total_orders, COUNT(DISTINCT o.order_id)::NUMERIC / COUNT(DISTINCT s.website_session_id) AS conversion_rate FROM toystore.website_sessions s LEFT JOIN toystore.orders o ON s.website_session_id = o.website_session_id GROUP BY s.is_repeat_session
```
